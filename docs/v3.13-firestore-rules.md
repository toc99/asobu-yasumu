# v3.13 Firestore Rules patch（本番適用前に必須）

重要: GitHub に現在の本番 Firestore Rules がないため、以下の断片を **現在デプロイ中のルールにマージ**してください。古いルール全体を上書きしないでください。このパッチを適用するまでは新機能の書き込みは permission-denied になります。

## 1. functions を match /databases/{database}/documents 内に追加

~~~javascript
function validKidRetroStart(h,playerId) {
  let prev=resource.data;let next=request.resource.data;let now=request.time.toMillis();
  return prev.activeStart==0 && prev.get('videoStartedAt',null)==null
    && prev.restUntil<=now && prev.get('manualRestUntil',0)<=now
    && next.diff(prev).affectedKeys().hasOnly(['activeStart','deviceId','startedAt','endedAt','updatedAt'])
    && next.activeStart is int && next.activeStart>=now-10800000 && next.activeStart<now
    && next.activeStart>=prev.restUntil && next.activeStart>=prev.get('manualRestUntil',0)
    && next.deviceId in ['sw1','sw2'] && next.startedAt==request.time
    && next.get('endedAt',null)==null && next.updatedAt==request.time;
}
function validKidRetroCorrection(h,playerId) {
  let prev=resource.data;let next=request.resource.data;let now=request.time.toMillis();
  return prev.activeStart is int && prev.activeStart>0
    && prev.get('videoStartedAt',null)==null
    && next.diff(prev).affectedKeys().hasOnly(['activeStart','updatedAt'])
    && next.activeStart is int && next.activeStart>=now-10800000
    && next.activeStart<prev.activeStart && next.activeStart>=prev.restUntil
    && next.activeStart>=prev.get('manualRestUntil',0)
    && next.updatedAt==request.time;
}
function releasePath(h,rid) {
  return /databases/$(database)/documents/households/$(h)/pokemonReleases/$(rid);
}
function validParentPokemonReleaseReward(h,rewardId) {
  let prev=resource.data;let next=request.resource.data;
  let rid=next.get('releaseEventId','');
  return isParent(h)
    && prev.get('revoked',false)!=true
    && prev.childId in ['older','younger']
    && next.diff(prev).affectedKeys().hasOnly(['revoked','revokedReason','revokedAt','revokedBy','releaseEventId'])
    && next.revoked==true && next.revokedReason=='released'
    && next.revokedAt==request.time && next.revokedBy==request.auth.uid
    && rid is string && rid.size()>0 && rid.size()<=180
    && existsAfter(releasePath(h,rid))
    && getAfter(releasePath(h,rid)).data.childId==prev.childId
    && getAfter(releasePath(h,rid)).data.rewardIds.hasAny([rewardId]);
}
function releaseRewardIsLinked(h,eventId,id,rid) {
  return existsAfter(rewardPath(h,rid))
    && getAfter(rewardPath(h,rid)).data.childId==id
    && getAfter(rewardPath(h,rid)).data.revoked==true
    && getAfter(rewardPath(h,rid)).data.revokedReason=='released'
    && getAfter(rewardPath(h,rid)).data.releaseEventId==eventId
    && getAfter(rewardPath(h,rid)).data.revokedBy==request.auth.uid
    && getAfter(rewardPath(h,rid)).data.revokedAt==request.time;
}
function validParentPokemonReleaseEvent(h,eventId) {
  let d=request.resource.data;
  return isParent(h)
    && d.keys().hasAll(['childId','createdBy','createdAt','reason','rewardIds','pokemon','count'])
    && d.keys().hasOnly(['childId','createdBy','createdAt','reason','rewardIds','pokemon','count'])
    && d.childId in ['older','younger'] && d.createdBy==request.auth.uid
    && d.createdAt==request.time && d.reason=='rules_violation' && d.count==3
    && d.rewardIds is list && d.rewardIds.size()==3
    && d.rewardIds[0] is string && d.rewardIds[1] is string && d.rewardIds[2] is string
    && d.rewardIds[0]!=d.rewardIds[1] && d.rewardIds[0]!=d.rewardIds[2] && d.rewardIds[1]!=d.rewardIds[2]
    && d.pokemon is list && d.pokemon.size()==3
    && getAfter(profilePath(h,d.childId)).data.pokemonLastReleaseEventId==eventId
    && getAfter(profilePath(h,d.childId)).data.updatedAt==request.time
    && releaseRewardIsLinked(h,eventId,d.childId,d.rewardIds[0])
    && releaseRewardIsLinked(h,eventId,d.childId,d.rewardIds[1])
    && releaseRewardIsLinked(h,eventId,d.childId,d.rewardIds[2]);
}
~~~

## 2. match /households/{h} 配下の既存マッチを拡張

- profiles/{playerId}: 既存の allow update の isKid(h) 側に、下記の OR 条件を追加。保護者の既存 isParent(h) 条件はそのまま残す。

~~~javascript
validKidRetroStart(h,playerId) || validKidRetroCorrection(h,playerId)
~~~

- rewards/{rewardId}: 現在の allow create は変更しない。既存の allow update 条件がある場合は残し、次の関数を OR で追加。既存で allow update, delete: if false; の場合は分割する。

~~~javascript
allow update: if validParentPokemonReleaseReward(h,rewardId);
allow delete: if false;
~~~

- pokemonReleases: 新規追加。

~~~javascript
match /pokemonReleases/{eventId} {
  allow read: if isMember(h);
  allow create: if validParentPokemonReleaseEvent(h,eventId);
  allow update, delete: if false;
}
~~~

- validV34Audit(h) の既存 action 列挙に追加（既存アクションとズル確認・解除などを削除しない）。
  - kid: start_backdated と start_corrected を auditActionTouchesProfile(h,d) 側に追加
  - parent: start_backdated、start_corrected、pokemon_release を auditActionTouchesProfile(h,d) 側に追加

## 3. セキュリティ・回帰テスト

- 子どもの遡及は3時間以内。未来時刻や休憩中の開始を拒否。
- 保護者の遡及は24時間以内。古いセッションには触らない。
- 子どもによる pokemonReleases 作成と rewards.revoked 更新は拒否。
- 逃がす操作は3件の reward 更新、イベント作成、profile更新、audit作成が一つのトランザクションで成功する。
- iPad に逃げた3匹の名前が通知される（アプリが開いている時、または次回オンラインで開いた時）。
- 旧バージョンのゲーム終了、休憩、図鑑、活動報酬、ズル判定に退行がない。

本番への Rules 適用前に Firebase Emulator などで権限の正負ケースをテストすること。画面だけをデプロイしない。
